# name : source : entry : profile : comma-separated defines

set(_mat_shader_dir ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
set(_mat_gen_dir ${CMAKE_CURRENT_BINARY_DIR}/generated/alamo_materials_spirv)
file(MAKE_DIRECTORY ${_mat_gen_dir})

set(_mat_dxc "${ALAMO_DXC}")
if(NOT EXISTS "${_mat_dxc}")
  find_program(_mat_dxc_found NAMES dxc)
  if(_mat_dxc_found)
    set(_mat_dxc "${_mat_dxc_found}")
  else()
    message(FATAL_ERROR "materials: DXC not found (ALAMO_DXC='${ALAMO_DXC}'); run toolchain\\3_renderer_deps.cmd")
  endif()
endif()

file(GLOB _mat_shader_sources CONFIGURE_DEPENDS ${_mat_shader_dir}/*.hlsl ${_mat_shader_dir}/*.hlsli)
set(_mat_headers "")
set(_mat_inc "// Generated by engine/materials/CMakeLists.txt. Do not edit.\n")
foreach(entry IN LISTS ALAMO_MATERIAL_PROGRAMS)
  string(REPLACE ":" ";" parts "${entry}")
  list(GET parts 0 name)
  list(GET parts 1 src)
  list(GET parts 2 ep)
  list(GET parts 3 profile)
  list(LENGTH parts nparts)
  set(defs "")
  set(defstr "")
  if(nparts GREATER 4)
    list(GET parts 4 defstr)
    if(NOT defstr STREQUAL "")
      string(REPLACE "," ";" deflist "${defstr}")
      foreach(d IN LISTS deflist)
        list(APPEND defs "-D" "${d}")
      endforeach()
    endif()
  endif()
  set(spv ${_mat_gen_dir}/${name}.spv)
  set(hdr ${_mat_gen_dir}/${name}.h)
  add_custom_command(
    OUTPUT ${spv}
    COMMAND "${_mat_dxc}" -nologo -spirv -fvk-use-dx-layout -fspv-target-env=vulkan1.3 -WX -O3
            -T ${profile} -E ${ep} ${defs} -I "${_mat_shader_dir}" -Fo "${spv}" "${_mat_shader_dir}/${src}"
    DEPENDS ${_mat_shader_sources}
    COMMENT "DXC ${name} (${src} ${ep} ${profile} ${defstr})"
    VERBATIM)
  add_custom_command(
    OUTPUT ${hdr}
    COMMAND ${CMAKE_COMMAND} -DIN=${spv} -DOUT=${hdr} -DSYM=spv_${name}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/spv_to_header.cmake
    DEPENDS ${spv} ${CMAKE_CURRENT_SOURCE_DIR}/cmake/spv_to_header.cmake
    VERBATIM)
  list(APPEND _mat_headers ${hdr})
  if(profile MATCHES "^vs")
    set(stage "Vertex")
  else()
    set(stage "Pixel")
  endif()
  string(APPEND _mat_inc "#include \"${name}.h\"\n")
  string(APPEND _mat_inc "ALAMO_SPV(${name}, \"${src}\", \"${ep}\", ${stage}, \"${defstr}\")\n")
endforeach()
file(CONFIGURE OUTPUT ${_mat_gen_dir}/programs.inc CONTENT "${_mat_inc}")
add_custom_target(alamo_materials_spirv DEPENDS ${_mat_headers})

# ---------------------------------------------------------------- library
add_library(alamo_materials STATIC
  src/d3d9_state.cpp
  src/pso.cpp
  src/params.cpp
  src/fxlc.cpp
  src/register_file.cpp
  src/fixed_function.cpp
  src/programs.cpp
  src/effect_loader.cpp
  src/translated.cpp
  ${_mat_headers}
)
add_library(alamo::materials ALIAS alamo_materials)
add_dependencies(alamo_materials alamo_materials_spirv)
target_include_directories(alamo_materials PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(alamo_materials PRIVATE ${_mat_gen_dir})

# build without it (build_module.cmd materials = core;materials) pulls it in here.

  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../shaderxlat ${CMAKE_BINARY_DIR}/engine/shaderxlat)
endif()
target_link_libraries(alamo_materials PUBLIC alamo_core alamo_fx2 alamo_shaderxlat)
