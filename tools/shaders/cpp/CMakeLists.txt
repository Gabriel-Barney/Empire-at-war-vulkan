cmake_minimum_required(VERSION 3.16)
project(alamo_fx2 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(alamo_fx2 STATIC fx2_reader.cpp)
target_include_directories(alamo_fx2 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  set(ALAMO_FX2_WARNINGS /W4)
else()
  set(ALAMO_FX2_WARNINGS -Wall -Wextra -Wpedantic)
endif()
target_compile_options(alamo_fx2 PRIVATE ${ALAMO_FX2_WARNINGS})
add_executable(fx2dump fx2dump.cpp)
target_link_libraries(fx2dump PRIVATE alamo_fx2)
target_compile_options(fx2dump PRIVATE ${ALAMO_FX2_WARNINGS})
